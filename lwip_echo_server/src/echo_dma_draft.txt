/* ============================================================================
 * echo_dma_draft  --  DMA-based tokenizer transfer (optimization R2)
 * ----------------------------------------------------------------------------
 * DRAFT / REFERENCE. This is NOT compiled (it's a .txt). When the DMA hardware
 * bitstream is built and you've re-read the .xsa in Vitis (so the BSP includes
 * the XAxiDma driver), integrate the pieces below into echo.c -- replacing the
 * byte-by-byte AXI-Lite send/drain (tok_send_byte loop + the pipeline_busy
 * drain) with the DMA path.
 *
 * Hardware this assumes (the build you're generating now):
 *   - AXI DMA in the block design, Simple mode, MM2S stream = 8-bit,
 *     S2MM stream = 16-bit; MM2S -> tokenizer s_axis, tokenizer m_axis -> S2MM.
 *   - The tokenizer's 0x0C TOKEN_COUNT register: write any value = clear,
 *     read = number of tokens produced since the last clear. (Simple-mode S2MM
 *     does not report received length, so we read the count from the IP.)
 *
 * CRITICAL: the DMA buffers (dma_in_buf, dma_tok_buf) MUST live in DDR -- the
 * AXI DMA's address window is the MIG range (0x8000_0000..0x9FFF_FFFF), so the
 * linker must place these in DDR, not local BRAM. If your .elf runs from BRAM,
 * put these buffers in a DDR section (e.g. via the linker script / a section
 * attribute) or malloc them from a DDR heap.
 * ============================================================================ */

#include "xaxidma.h"
#include "xil_cache.h"
#include "xil_io.h"
#include "xparameters.h"

/* ---- locked from this build's xparameters.h (SDT / base-address flow) ---- */
/* Tokenizer AXI-Lite base = XPAR_TOKENIZER_AXI_LITE_0_BASEADDR (matches the
 * UART banner "Tokenizer HW base address: 0x44A00000"). */
#define TOKENIZER_BASE      0x44A00000u
/* OUR AXI DMA base = XPAR_AXI_DMA_0_BASEADDR. NOTE: there are TWO DMAs in this
 * design -- ours is 0x41E10000; the AXI Ethernet's own DMA is 0x41E00000.
 * Do NOT use the ethernet one. In the SDT flow LookupConfig takes the base
 * address, not a device id. */
#define TOK_DMA_BASEADDR    XPAR_AXI_DMA_0_BASEADDR   /* = 0x41E10000 */
/* ------------------------------------------------------------------------- */

#define TOK_STATUS_REG   (TOKENIZER_BASE + 0x08u)  /* bit3 pipeline_busy, bit1 token avail, ... */
#define TOK_COUNT_REG    (TOKENIZER_BASE + 0x0Cu)  /* R2 TOKEN_COUNT (write=clear, read=count) */

#define MAX_INPUT_BYTES  2048u      /* largest text chunk streamed in one DMA transfer */
#define MAX_TOKENS       1024u      /* largest token response (each token = 2 bytes) */

/* DMA buffers -- 64-byte (cache-line) aligned, and they MUST resolve to DDR. */
static u8  dma_in_buf [MAX_INPUT_BYTES] __attribute__((aligned(64)));
static u16 dma_tok_buf[MAX_TOKENS]      __attribute__((aligned(64)));

static XAxiDma axi_dma;

/* ---- call once at startup (e.g. right after the tokenizer banner) -------- */
int tokenizer_dma_init(void)
{
    XAxiDma_Config *cfg = XAxiDma_LookupConfig(TOK_DMA_BASEADDR);
    if (cfg == NULL) {
        xil_printf("DMA: LookupConfig failed\r\n");
        return XST_FAILURE;
    }
    if (XAxiDma_CfgInitialize(&axi_dma, cfg) != XST_SUCCESS) {
        xil_printf("DMA: CfgInitialize failed\r\n");
        return XST_FAILURE;
    }
    if (XAxiDma_HasSg(&axi_dma)) {           /* we configured Simple mode -- SG must be off */
        xil_printf("DMA: unexpected Scatter-Gather mode\r\n");
        return XST_FAILURE;
    }
    XAxiDma_IntrDisable(&axi_dma, XAXIDMA_IRQ_ALL_MASK, XAXIDMA_DEVICE_TO_DMA);
    XAxiDma_IntrDisable(&axi_dma, XAXIDMA_IRQ_ALL_MASK, XAXIDMA_DMA_TO_DEVICE);
    return XST_SUCCESS;
}

/* ---- tokenize `len` bytes already placed in dma_in_buf ------------------- *
 * The input MUST end on a word boundary (whitespace/newline) so the final word
 * flushes and the tokenizer asserts m_axis TLAST on its last token -- otherwise
 * the S2MM transfer never sees TLAST and the wait below would hang. The caller
 * (recv_callback) guarantees this by appending '\n' if needed.
 * Returns the number of tokens (now sitting in dma_tok_buf), or -1 on error. */
static int tokenizer_dma_run(u32 len)
{
    u32 timeout;

    if (len == 0u || len > MAX_INPUT_BYTES) return -1;

    /* clear the hardware token counter for this transfer */
    Xil_Out32(TOK_COUNT_REG, 0u);

    /* push the input bytes out to DDR; reserve/clean the output region */
    Xil_DCacheFlushRange((UINTPTR)dma_in_buf, len);
    Xil_DCacheInvalidateRange((UINTPTR)dma_tok_buf, MAX_TOKENS * sizeof(u16));

    /* arm the receive (S2MM) FIRST so it's ready before any token is produced */
    if (XAxiDma_SimpleTransfer(&axi_dma, (UINTPTR)dma_tok_buf,
            MAX_TOKENS * sizeof(u16), XAXIDMA_DEVICE_TO_DMA) != XST_SUCCESS)
        return -1;
    /* then kick the send (MM2S): streams the bytes into the tokenizer */
    if (XAxiDma_SimpleTransfer(&axi_dma, (UINTPTR)dma_in_buf,
            len, XAXIDMA_DMA_TO_DEVICE) != XST_SUCCESS)
        return -1;

    /* wait for input to finish streaming in */
    timeout = 100000000u;
    while (XAxiDma_Busy(&axi_dma, XAXIDMA_DMA_TO_DEVICE) && --timeout) { }
    if (timeout == 0u) { xil_printf("DMA: MM2S timeout\r\n"); return -1; }

    /* wait for all tokens to stream back; S2MM completes when the IP asserts TLAST */
    timeout = 100000000u;
    while (XAxiDma_Busy(&axi_dma, XAXIDMA_DEVICE_TO_DMA) && --timeout) { }
    if (timeout == 0u) { xil_printf("DMA: S2MM timeout (no TLAST? input not boundary-terminated)\r\n"); return -1; }

    /* tokens are in DDR now -- invalidate so the CPU reads fresh data */
    Xil_DCacheInvalidateRange((UINTPTR)dma_tok_buf, MAX_TOKENS * sizeof(u16));

    /* the IP's 0x0C register tells us how many tokens it produced */
    return (int)(Xil_In32(TOK_COUNT_REG) & 0xFFFFu);
}

/* ---- recv_callback (replaces the byte-banging version in echo.c) -------- *
 * Streams the received text through the tokenizer via DMA and writes the token
 * IDs back as decimal text. NOTE: this draft treats each TCP segment as self-
 * contained and appends a newline if it doesn't already end on whitespace, so a
 * word split across two TCP segments is NOT reassembled (that was M3's job). For
 * the DMA proof-of-concept that's fine; reassembly can be added later by
 * buffering until a boundary before launching the DMA. */
static err_t recv_callback(void *arg, struct tcp_pcb *tpcb,
                           struct pbuf *p, err_t err)
{
    u32 len = 0u;
    struct pbuf *q;
    int ntok, i, n;
    char out[16];

    if (p == NULL) {                    /* remote closed */
        tcp_close(tpcb);
        return ERR_OK;
    }

    /* copy the pbuf chain into the DDR input buffer */
    for (q = p; q != NULL && len < MAX_INPUT_BYTES; q = q->next) {
        u16 c = q->len;
        if (len + c > MAX_INPUT_BYTES) c = (u16)(MAX_INPUT_BYTES - len);
        memcpy(&dma_in_buf[len], q->payload, c);
        len += c;
    }

    /* guarantee a trailing word boundary so the last word flushes (-> TLAST) */
    {
        u8 last = (len > 0u) ? dma_in_buf[len - 1u] : (u8)'\n';
        int is_ws = (last == ' ' || last == '\t' || last == '\r' || last == '\n');
        if (!is_ws && len < MAX_INPUT_BYTES) dma_in_buf[len++] = (u8)'\n';
    }

    ntok = tokenizer_dma_run(len);

    if (ntok > 0) {
        for (i = 0; i < ntok; i++) {
            n = snprintf(out, sizeof(out), (i + 1 < ntok) ? "%u " : "%u\r\n",
                         (unsigned)dma_tok_buf[i]);
            if (n > 0) tcp_write(tpcb, out, (u16)n, TCP_WRITE_FLAG_COPY);
        }
        tcp_output(tpcb);
    }

    tcp_recved(tpcb, p->tot_len);
    pbuf_free(p);
    return ERR_OK;
}
